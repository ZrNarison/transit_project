class Materiaux(models.Model):
    libelle = models.CharField(
        max_length=100,
        verbose_name="LibellÃ©",
    )

    unite = models.CharField(
        max_length=50,
        verbose_name="UnitÃ©",
    )

    class Meta:
        ordering = ["libelle"]
        verbose_name = "MatÃ©riau"
        verbose_name_plural = "MatÃ©riaux"

    def __str__(self):
        return f"{self.libelle} ({self.unite})"

    @property
    def total_entrees(self):
        return (
            self.entrees.aggregate(
                total=Sum("entree")
            )["total"]
            or Decimal("0")
        )

    @property
    def total_divisions(self):
        return (
            MateriauxDivision.objects.filter(
                entree__materiau=self
            ).aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    @property
    def total_sorties(self):
        return (
            MateriauxOut.objects.filter(
                division__entree__materiau=self
            ).aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    @property
    def total_retours(self):
        return (
            MateriauxRetour.objects.filter(
                sortie__division__entree__materiau=self
            ).aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    @property
    def stock_entrepot(self):
        """
        Stock encore disponible dans l'entrepÃ´t,
        avant division vers le magasin.
        """
        return self.total_entrees - self.total_divisions

    @property
    def stock_magasin(self):
        """
        Stock magasin :

        divisions
        - sorties vers chantier
        + retours du chantier
        """
        return (
            self.total_divisions
            - self.total_sorties
            + self.total_retours
        )

    @property
    def stock_total(self):
        return self.stock_entrepot + self.stock_magasin

class MateriauxDivision(models.Model):
    entree = models.ForeignKey(
        MateriauxEntree,
        on_delete=models.PROTECT,
        related_name="divisions",
        verbose_name="EntrÃ©e entrepÃ´t",
    )

    destination = models.CharField(
        max_length=255,
        verbose_name="Destination / Magasin",
    )

    quantite = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        verbose_name="QuantitÃ© divisÃ©e",
    )

    date_division = models.DateField(
        verbose_name="Date de division",
    )

    enregistreur = models.ForeignKey(
        AppUser,
        on_delete=models.PROTECT,
        related_name="materiaux_divisions",
        verbose_name="EnregistrÃ© par",
    )

    date_creation = models.DateTimeField(
        auto_now_add=True,
    )

    class Meta:
        ordering = ["-date_division", "-id"]
        verbose_name = "Division matÃ©riau"
        verbose_name_plural = "Divisions matÃ©riaux"

    def clean(self):
        if not self.entree_id:
            return

        if self.quantite is None or self.quantite <= 0:
            raise ValidationError({
                "quantite": (
                    "La quantitÃ© doit Ãªtre supÃ©rieure Ã  zÃ©ro."
                )
            })

        divisions_existantes = (
            self.entree.divisions
            .exclude(pk=self.pk)
            .aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

        disponible = self.entree.entree - divisions_existantes

        if self.quantite > disponible:
            raise ValidationError({
                "quantite": (
                    f"QuantitÃ© insuffisante dans l'entrepÃ´t. "
                    f"Disponible : {disponible} "
                    f"{self.entree.materiau.unite}."
                )
            })

        if self.pk:
            sorties = (
                self.sorties
                .aggregate(
                    total=Sum("quantite")
                )["total"]
                or Decimal("0")
            )

            if self.quantite < sorties:
                raise ValidationError({
                    "quantite": (
                        "Impossible de rÃ©duire cette division en dessous "
                        "des quantitÃ©s dÃ©jÃ  sorties."
                    )
                })

    def save(self, *args, **kwargs):
        self.full_clean()
        return super().save(*args, **kwargs)

    @property
    def quantite_sortie(self):
        return (
            self.sorties.aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    @property
    def quantite_retour(self):
        return (
            MateriauxRetour.objects.filter(
                sortie__division=self
            ).aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    @property
    def stock_restant(self):
        return (
            self.quantite
            - self.quantite_sortie
            + self.quantite_retour
        )

    def __str__(self):
        return (
            f"{self.entree.materiau.libelle} - "
            f"{self.destination} - "
            f"{self.quantite} "
            f"{self.entree.materiau.unite}"
        )

    vehicule = models.CharField(
        max_length=100,
        blank=True,
        verbose_name="VÃ©hicule",
    )

    # --------------------------------------------------------
    # NOUVELLES RELATIONS PROJET
    # --------------------------------------------------------

    projet = models.ForeignKey(
        "projet.Projet",
        on_delete=models.PROTECT,
        related_name="sorties_materiaux",

        blank=True,
        verbose_name="Projet",

    point_projet = models.ForeignKey(
        "projet.PointProjet",
        on_delete=models.PROTECT,
        related_name="sorties_materiaux",

        blank=True,
        verbose_name="Point de chantier",
    )

    enregistreur = models.ForeignKey(
        AppUser,
        on_delete=models.PROTECT,
        related_name="materiaux_sorties",
        verbose_name="EnregistrÃ© par",
    )

    date_creation = models.DateTimeField(
        auto_now_add=True,
    )

    class Meta:
        ordering = ["-date_sortie", "-id"]
        verbose_name = "Sortie matÃ©riau"
        verbose_name_plural = "Sorties matÃ©riaux"

    def clean(self):
        if not self.division_id:
            return

        if self.quantite is None or self.quantite <= 0:
            raise ValidationError({
                "quantite": (
                    "La quantitÃ© doit Ãªtre supÃ©rieure Ã  zÃ©ro."
                )
            })

        sorties_existantes = (
            self.division.sorties
            .exclude(pk=self.pk)
            .aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

        retours_existants = (
            MateriauxRetour.objects.filter(
                sortie__division=self.division
            )
            .aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

        disponible = (
            self.division.quantite
            - sorties_existantes
            + retours_existants
        )

        if self.quantite > disponible:
            raise ValidationError({
                "quantite": (
                    f"Stock magasin insuffisant. "
                    f"Disponible : {disponible} "
                    f"{self.division.entree.materiau.unite}."
                )
            })

        # Le point doit appartenir au mÃªme projet.
        if self.point_projet_id and self.projet_id:
            if self.point_projet.projet_id != self.projet_id:
                raise ValidationError({
                    "point_projet": (
                        "Le point de chantier doit appartenir "
                        "au projet sÃ©lectionnÃ©."
                    )
                })

    def save(self, *args, **kwargs):
        self.full_clean()
        return super().save(*args, **kwargs)

    @property
    def quantite_transportee(self):
        return (
            self.transports.filter(
                type_transport=ActiviteTransport.MATERIAUX
            ).aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    @property
    def quantite_non_transportee(self):
        return max(
            self.quantite - self.quantite_transportee,
            Decimal("0"),
        )

    @property
    def quantite_retournee(self):
        return (
            self.retours.aggregate(
                total=Sum("quantite")
            )["total"]
            or Decimal("0")
        )

    def __str__(self):
        return (
            f"{self.division.entree.materiau.libelle} - "
            f"{self.quantite} "
            f"{self.division.entree.materiau.unite}"
        )

        if self.quantite is not None and self.quantite < 0:
            errors["quantite"] = (
                "La quantitÃ© ne peut pas Ãªtre nÃ©gative."
            )

        if (
            self.kilometrage_depart is not None
            and self.kilometrage_depart < 0
        ):
            errors["kilometrage_depart"] = (
                "Le kilomÃ©trage de dÃ©part ne peut pas Ãªtre nÃ©gatif."
            )

        if (
            self.kilometrage_arrivee is not None
            and self.kilometrage_arrivee < 0
        ):
            errors["kilometrage_arrivee"] = (
                "Le kilomÃ©trage d'arrivÃ©e ne peut pas Ãªtre nÃ©gatif."
            )

        if (
            self.kilometrage_arrivee is not None
            and self.kilometrage_depart is not None
            and self.kilometrage_arrivee < self.kilometrage_depart
        ):
            errors["kilometrage_arrivee"] = (
                "Le kilomÃ©trage d'arrivÃ©e doit Ãªtre supÃ©rieur "
                "ou Ã©gal au kilomÃ©trage de dÃ©part."
            )

        # ----------------------------------------------------
        # Un transport de matÃ©riaux doit Ãªtre liÃ© Ã  une sortie.
        # ----------------------------------------------------

        if (
            self.type_transport == self.MATERIAUX
            and self.sortie_materiau_id
        ):
            sortie = self.sortie_materiau

            transports_existants = (
                sortie.transports
                .exclude(pk=self.pk)
                .filter(
                    type_transport=self.MATERIAUX
                )
                .aggregate(
                    total=Sum("quantite")
                )["total"]
                or Decimal("0")
            )

            disponible = (
                sortie.quantite
                - transports_existants
            )

            if self.quantite > disponible:
                errors["quantite"] = (
                    f"QuantitÃ© trop Ã©levÃ©e. "
                    f"Disponible pour transport : "
                    f"{disponible} "
                    f"{sortie.division.entree.materiau.unite}."
                )

        # ----------------------------------------------------
        # CohÃ©rence Projet / Point
        # ----------------------------------------------------

        if self.point_projet_id and self.projet_id:
            if self.point_projet.projet_id != self.projet_id:
                errors["point_projet"] = (
                    "Le point sÃ©lectionnÃ© n'appartient pas "
                    "au projet sÃ©lectionnÃ©."
                )

        # ----------------------------------------------------
        # CohÃ©rence Projet / Sortie matÃ©riau
        # ----------------------------------------------------

        if self.sortie_materiau_id and self.projet_id:
            if (
                self.sortie_materiau.projet_id
                and self.sortie_materiau.projet_id != self.projet_id
            ):
                errors["sortie_materiau"] = (
                    "La sortie de matÃ©riau appartient Ã  un autre projet."
                )

        if errors:
            raise ValidationError(errors)

    def save(self, *args, **kwargs):
        self.full_clean()
        return super().save(*args, **kwargs)

    @property
    def distance_km(self):
        return max(
            (
                self.kilometrage_arrivee
                - self.kilometrage_depart
            ),
            Decimal("0"),
        )

    @property
    def materiau(self):
        if self.sortie_materiau_id:
            return self.sortie_materiau.division.entree.materiau
        return None

    def __str__(self):
        designation = (
            self.mission

        )

        return (
            f"{self.date_activite} - "
            f"{self.vehicule.immatriculation} - "
            f"{designation}"
        )
